expx = FALSE

tt <- readODS::read_ods("datas/HDJ-TO.ods") |> 
  mutate(across(is.character, as.factor)) |> 
  mutate(across(Comprehension_verifiee:Comprehension_verifiee, ~fct_relevel(.x,
    "Toujours", "Parfois", "Rarement", "Jamais"
  ))) |> 
    mutate(Evaluation_fragilite = fct_relevel(Evaluation_fragilite,
    "Toujours", "Parfois"
  )) |> 
  mutate(across(starts_with("Recours"),~ fct_relevel(.x,                                                     "Systématique",
  "Selon situation",
  "Rarement", "Jamais"))) |> 
  mutate(across(starts_with("Ancien"),~ fct_relevel(.x, "0-5ans", "6-10ans",
                                                    ">10ans"))) |> 
  mutate(Temps_information = fct_relevel(Temps_information, 
                                         "10-20min", ">20min")) |> 
  mutate(Acces_soins_support_suffisant = fct_relevel(Acces_soins_support_suffisant, 
                                         "Oui","Partiellement" ,"Non"))
